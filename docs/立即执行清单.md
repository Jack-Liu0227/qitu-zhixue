### Agent 3: 数据库迁移

```bash
# 1. 创建迁移目录（如果尚未创建）
mkdir -p migrations/{data,rollback}

# 2. 创建 AI 搭档表（第二个迁移文件）
cd migrations
cat > 002_ai_tutor_tables.sql << EOF
-- AI 搭档核心表
-- 版本：v1.0

-- AI 会话表
CREATE TABLE ai_tutor_sessions (
    session_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES students(student_id),
    project_id UUID REFERENCES student_projects(project_id),
    stage VARCHAR(50),
    tutor_id VARCHAR(50) NOT NULL,
    context JSONB NOT NULL DEFAULT '{}',
    state VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    last_active_at TIMESTAMPTZ DEFAULT NOW(),
    completed_at TIMESTAMPTZ,
    INDEX idx_student_active (student_id, state),
    INDEX idx_project (project_id)
);

-- AI 消息表
CREATE TABLE ai_tutor_messages (
    message_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES ai_tutor_sessions(session_id) ON DELETE CASCADE,
    role VARCHAR(20) NOT NULL,
    content TEXT NOT NULL,
    content_type VARCHAR(20) DEFAULT 'text',
    hint_level INT CHECK (hint_level BETWEEN 1 AND 5),
    stage VARCHAR(50),
    capability_used VARCHAR(50),
    tool_calls JSONB,
    attachments JSONB,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    INDEX idx_session (session_id, created_at)
);

-- 三层记忆 L1
CREATE TABLE memory_l1_entities (
    entity_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES students(student_id),
    entity_type VARCHAR(50) NOT NULL,
    entity_ref VARCHAR(255) NOT NULL,
    fingerprint TEXT,
    snapshot JSONB NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    INDEX idx_student_type (student_id, entity_type),
    UNIQUE (entity_ref)
);

-- 三层记忆 L2
CREATE TABLE memory_l2_summaries (
    summary_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES students(student_id),
    scope VARCHAR(50) NOT NULL,
    scope_ref VARCHAR(255) NOT NULL,
    facts JSONB NOT NULL,
    source_entities UUID[] NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    INDEX idx_student_scope (student_id, scope, scope_ref)
);

-- 三层记忆 L3
CREATE TABLE memory_l3_profiles (
    profile_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL UNIQUE REFERENCES students(student_id),
    profile_data JSONB NOT NULL,
    evidence_chains JSONB NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 人工介入工单
CREATE TABLE intervention_requests (
    request_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES students(student_id),
    project_id UUID REFERENCES student_projects(project_id),
    task_id UUID,
    session_id UUID REFERENCES ai_tutor_sessions(session_id),
    type VARCHAR(50) NOT NULL,
    reason TEXT NOT NULL,
    suggested_handler VARCHAR(50),
    urgency VARCHAR(20) NOT NULL,
    blocking BOOLEAN DEFAULT FALSE,
    ai_context JSONB NOT NULL,
    evidence_snippets JSONB,
    status VARCHAR(20) DEFAULT 'pending',
    assigned_to UUID REFERENCES users(user_id),
    assigned_at TIMESTAMPTZ,
    resolution_notes TEXT,
    resolved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    INDEX idx_student_status (student_id, status),
    INDEX idx_assigned (assigned_to, status),
    INDEX idx_urgency (urgency, created_at)
);

COMMENT ON TABLE ai_tutor_sessions IS 'AI 搭档会话记录';
COMMENT ON TABLE ai_tutor_messages IS 'AI 对话消息';
COMMENT ON TABLE memory_l1_entities IS '三层记忆 L1：实体镜像';
COMMENT ON TABLE memory_l2_summaries IS '三层记忆 L2：阶段摘要';
COMMENT ON TABLE memory_l3_profiles IS '三层记忆 L3：综合画像';
COMMENT ON TABLE intervention_requests IS '人工介入工单';
EOF

# 3. 执行迁移
psql -U qitu_admin -d qitu_zhixue -f 002_ai_tutor_tables.sql
```

## 🆘 常见问题

### Q1: pnpm 在网络共享路径报错？
**A**: 已在 `.npmrc` 中配置 `store-dir=C:\\.pnpm-store`，重新运行 `pnpm install`

### Q2: PostgreSQL 连接被拒绝？
**A**: 检查 `pg_hba.conf`，确保有 `host all all 127.0.0.1/32 md5`

### Q3: Python 虚拟环境激活失败？
**A**: Windows 需要执行 `Set-ExecutionPolicy RemoteSigned -Scope CurrentUser`

### Q4: Redis 启动失败？
**A**: 端口 6379 被占用，使用 `redis-server --port 6380` 更改端口

### Q5: FastAPI 启动后前端无法访问？
**A**: 检查 CORS 配置，确保 `allow_origins` 包含前端地址
