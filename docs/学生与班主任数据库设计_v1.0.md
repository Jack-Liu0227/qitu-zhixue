## 📋 核心业务规则

### ✅ 已明确的约束（AGENTS.md）

1. **一个学生同一时间只能有一个当前班主任**
2. **班主任只能查看已分配给自己的学生**
3. **学生未确认意图时不得创建正式项目**
4. **项目状态转换、AI 决策、成长档案只能由服务端完成**
5. **所有写操作考虑幂等性**

### 4. 班主任分配表（核心表）

```sql
-- ═══════════════════════════════════════════════════════
-- 班主任分配表（学生 ↔ 班主任关系）
-- 
-- 核心约束：一个学生同一时间只能有一个当前班主任
-- ═══════════════════════════════════════════════════════
CREATE TABLE mentor_assignments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE RESTRICT,
  teacher_id UUID NOT NULL REFERENCES teachers(id) ON DELETE RESTRICT,
  
  -- 分配时间
  assigned_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  assigned_by UUID REFERENCES users(id),
    -- 谁分配的（管理员 user_id）
  
  -- 结束时间（NULL 表示当前班主任）
  ended_at TIMESTAMP WITH TIME ZONE,
  end_reason VARCHAR(50),
    -- transferred | student_graduated | teacher_resigned | manual_change
  ended_by UUID REFERENCES users(id),
  
  -- 状态
  is_current BOOLEAN NOT NULL DEFAULT TRUE,
  
  -- 备注
  notes TEXT,
  
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  
  -- ✅ 核心约束：一个学生同一时间只能有一个当前班主任
  CONSTRAINT unique_current_mentor_per_student 
    EXCLUDE USING gist (
      student_id WITH =,
      tstzrange(assigned_at, ended_at, '[)') WITH &&
    ) WHERE (is_current = TRUE),
  
  -- 结束时间必须晚于开始时间
  CONSTRAINT check_end_after_start CHECK (
    ended_at IS NULL OR ended_at > assigned_at
  ),
  
  -- 如果有结束时间，必须有结束原因
  CONSTRAINT check_end_reason CHECK (
    (ended_at IS NULL AND end_reason IS NULL) OR
    (ended_at IS NOT NULL AND end_reason IS NOT NULL)
  )
);

-- 索引
CREATE INDEX idx_mentor_assignments_student ON mentor_assignments(student_id);
CREATE INDEX idx_mentor_assignments_teacher ON mentor_assignments(teacher_id);
CREATE INDEX idx_mentor_assignments_current 
  ON mentor_assignments(student_id, is_current) 
  WHERE is_current = TRUE;
CREATE INDEX idx_mentor_assignments_date_range 
  ON mentor_assignments USING gist (tstzrange(assigned_at, ended_at, '[)'));

CREATE TRIGGER mentor_assignments_updated_at
  BEFORE UPDATE ON mentor_assignments
  FOR EACH ROW
  EXECUTE FUNCTION update_updated_at();

-- ═══════════════════════════════════════════════════════
-- 触发器：自动更新班主任的 current_student_count
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION update_teacher_student_count()
RETURNS TRIGGER AS $$
BEGIN
  -- 如果是新增或修改为 is_current = TRUE
  IF (TG_OP = 'INSERT' AND NEW.is_current = TRUE) OR
     (TG_OP = 'UPDATE' AND NEW.is_current = TRUE AND OLD.is_current = FALSE) THEN
    UPDATE teachers
    SET current_student_count = current_student_count + 1
    WHERE id = NEW.teacher_id;
  END IF;
  
  -- 如果是修改为 is_current = FALSE 或删除
  IF (TG_OP = 'UPDATE' AND NEW.is_current = FALSE AND OLD.is_current = TRUE) OR
     (TG_OP = 'DELETE' AND OLD.is_current = TRUE) THEN
    UPDATE teachers
    SET current_student_count = current_student_count - 1
    WHERE id = COALESCE(NEW.teacher_id, OLD.teacher_id);
  END IF;
  
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER mentor_assignments_update_count
  AFTER INSERT OR UPDATE OR DELETE ON mentor_assignments
  FOR EACH ROW
  EXECUTE FUNCTION update_teacher_student_count();
```

## 🔒 权限与安全规则

### Row Level Security（RLS）策略

```sql
-- ═══════════════════════════════════════════════════════
-- 启用 RLS
-- ═══════════════════════════════════════════════════════
ALTER TABLE students ENABLE ROW LEVEL SECURITY;
ALTER TABLE teachers ENABLE ROW LEVEL SECURITY;
ALTER TABLE mentor_assignments ENABLE ROW LEVEL SECURITY;
ALTER TABLE guardian_links ENABLE ROW LEVEL SECURITY;

-- ═══════════════════════════════════════════════════════
-- 学生只能查看自己的档案
-- ═══════════════════════════════════════════════════════
CREATE POLICY students_select_own
  ON students FOR SELECT
  USING (user_id = current_setting('app.user_id')::UUID);

-- ═══════════════════════════════════════════════════════
-- 班主任只能查看已分配给自己的学生
-- ═══════════════════════════════════════════════════════
CREATE POLICY students_select_by_teacher
  ON students FOR SELECT
  USING (
    id IN (
      SELECT ma.student_id
      FROM mentor_assignments ma
      JOIN teachers t ON t.id = ma.teacher_id
      WHERE t.user_id = current_setting('app.user_id')::UUID
        AND ma.is_current = TRUE
    )
  );

-- ═══════════════════════════════════════════════════════
-- 家长只能查看已授权的孩子
-- ═══════════════════════════════════════════════════════
CREATE POLICY students_select_by_parent
  ON students FOR SELECT
  USING (
    id IN (
      SELECT gl.student_id
      FROM guardian_links gl
      JOIN parents p ON p.id = gl.parent_id
      WHERE p.user_id = current_setting('app.user_id')::UUID
        AND gl.status = 'active'
        AND gl.can_view_progress = TRUE
    )
  );

-- ═══════════════════════════════════════════════════════
-- 管理员可以查看所有
-- ═══════════════════════════════════════════════════════
CREATE POLICY students_select_by_admin
  ON students FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM user_roles
      WHERE user_id = current_setting('app.user_id')::UUID
        AND role = 'admin'
    )
  );
```

## 📝 核心业务逻辑（存储过程）

### 1. 分配班主任（幂等操作）

```sql
-- ═══════════════════════════════════════════════════════
-- 为学生分配班主任
-- 
-- 功能：
--   1. 自动结束旧的分配关系
--   2. 创建新的分配关系
--   3. 更新班主任的学生计数
--   4. 幂等性：重复分配同一班主任不会报错
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION assign_mentor(
  p_student_id UUID,
  p_teacher_id UUID,
  p_assigned_by UUID,
  p_notes TEXT DEFAULT NULL
)
RETURNS UUID AS $$
DECLARE
  v_old_assignment_id UUID;
  v_new_assignment_id UUID;
BEGIN
  -- 1. 检查学生和班主任是否存在
  IF NOT EXISTS (SELECT 1 FROM students WHERE id = p_student_id) THEN
    RAISE EXCEPTION '学生不存在: %', p_student_id;
  END IF;
  
  IF NOT EXISTS (SELECT 1 FROM teachers WHERE id = p_teacher_id) THEN
    RAISE EXCEPTION '班主任不存在: %', p_teacher_id;
  END IF;
  
  -- 2. 检查班主任是否已满员
  IF EXISTS (
    SELECT 1 FROM teachers
    WHERE id = p_teacher_id
      AND current_student_count >= max_students
  ) THEN
    RAISE EXCEPTION '班主任 % 已达到最大学生数限制', p_teacher_id;
  END IF;
  
  -- 3. 查找当前分配关系
  SELECT id INTO v_old_assignment_id
  FROM mentor_assignments
  WHERE student_id = p_student_id
    AND is_current = TRUE
  LIMIT 1;
  
  -- 4. 幂等性检查：如果已经分配给该班主任，直接返回
  IF v_old_assignment_id IS NOT NULL THEN
    IF EXISTS (
      SELECT 1 FROM mentor_assignments
      WHERE id = v_old_assignment_id
        AND teacher_id = p_teacher_id
    ) THEN
      RAISE NOTICE '学生 % 已经分配给班主任 %，无需重复分配', 
        p_student_id, p_teacher_id;
      RETURN v_old_assignment_id;
    END IF;
  END IF;
  
  -- 5. 结束旧的分配关系
  IF v_old_assignment_id IS NOT NULL THEN
    UPDATE mentor_assignments
    SET 
      ended_at = NOW(),
      end_reason = 'transferred',
      ended_by = p_assigned_by,
      is_current = FALSE
    WHERE id = v_old_assignment_id;
    
    RAISE NOTICE '已结束旧的分配关系: %', v_old_assignment_id;
  END IF;
  
  -- 6. 创建新的分配关系
  INSERT INTO mentor_assignments (
    student_id,
    teacher_id,
    assigned_by,
    notes,
    is_current
  ) VALUES (
    p_student_id,
    p_teacher_id,
    p_assigned_by,
    p_notes,
    TRUE
  )
  RETURNING id INTO v_new_assignment_id;
  
  RAISE NOTICE '已创建新的分配关系: %', v_new_assignment_id;
  
  RETURN v_new_assignment_id;
END;
$$ LANGUAGE plpgsql;

-- 使用示例
-- SELECT assign_mentor(
--   '学生UUID'::UUID,
--   '班主任UUID'::UUID,
--   '管理员UUID'::UUID,
--   '首次分配'
-- );
```

### 2. 获取学生的当前班主任

```sql
-- ═══════════════════════════════════════════════════════
-- 获取学生的当前班主任信息
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION get_current_mentor(p_student_id UUID)
RETURNS TABLE (
  mentor_assignment_id UUID,
  teacher_id UUID,
  teacher_name VARCHAR,
  teacher_bio TEXT,
  assigned_at TIMESTAMP WITH TIME ZONE,
  days_assigned INT
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    ma.id AS mentor_assignment_id,
    t.id AS teacher_id,
    t.full_name AS teacher_name,
    t.bio AS teacher_bio,
    ma.assigned_at,
    EXTRACT(DAY FROM NOW() - ma.assigned_at)::INT AS days_assigned
  FROM mentor_assignments ma
  JOIN teachers t ON t.id = ma.teacher_id
  WHERE ma.student_id = p_student_id
    AND ma.is_current = TRUE
  LIMIT 1;
END;
$$ LANGUAGE plpgsql;
```

### 3. 获取班主任的所有当前学生

```sql
-- ═══════════════════════════════════════════════════════
-- 获取班主任负责的所有学生列表
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION get_teacher_students(p_teacher_id UUID)
RETURNS TABLE (
  student_id UUID,
  student_name VARCHAR,
  student_grade VARCHAR,
  assigned_at TIMESTAMP WITH TIME ZONE,
  days_assigned INT,
  active_projects_count BIGINT
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    s.id AS student_id,
    s.nickname AS student_name,
    s.grade_level AS student_grade,
    ma.assigned_at,
    EXTRACT(DAY FROM NOW() - ma.assigned_at)::INT AS days_assigned,
    COUNT(p.id) AS active_projects_count
  FROM mentor_assignments ma
  JOIN students s ON s.id = ma.student_id
  LEFT JOIN projects p ON p.student_id = s.id 
    AND p.status IN ('active', 'in_progress')
  WHERE ma.teacher_id = p_teacher_id
    AND ma.is_current = TRUE
  GROUP BY s.id, s.nickname, s.grade_level, ma.assigned_at
  ORDER BY ma.assigned_at DESC;
END;
$$ LANGUAGE plpgsql;

-- 使用示例
-- SELECT * FROM get_teacher_students('班主任UUID'::UUID);
```

## 📊 查询示例

### 1. 查询学生的完整档案（含班主任和家长）

```sql
SELECT 
  s.id AS student_id,
  s.nickname,
  s.grade_level,
  s.interests,
  
  -- 当前班主任
  t.full_name AS current_mentor_name,
  ma.assigned_at AS mentor_assigned_at,
  
  -- 家长信息
  ARRAY_AGG(
    DISTINCT jsonb_build_object(
      'parent_name', p.full_name,
      'relationship', gl.relationship
    )
  ) AS parents
  
FROM students s
LEFT JOIN mentor_assignments ma ON ma.student_id = s.id AND ma.is_current = TRUE
LEFT JOIN teachers t ON t.id = ma.teacher_id
LEFT JOIN guardian_links gl ON gl.student_id = s.id AND gl.status = 'active'
LEFT JOIN parents p ON p.id = gl.parent_id

WHERE s.id = '学生UUID'::UUID

GROUP BY s.id, s.nickname, s.grade_level, s.interests, 
         t.full_name, ma.assigned_at;
```

### 4. 查询班主任工作负载统计

```sql
SELECT 
  t.id,
  t.full_name,
  t.current_student_count,
  t.max_students,
  ROUND(
    (t.current_student_count::NUMERIC / t.max_students) * 100, 
    2
  ) AS workload_percentage,
  
  COUNT(DISTINCT p.id) AS total_active_projects,
  COUNT(DISTINCT i.id) AS pending_interventions
  
FROM teachers t
LEFT JOIN mentor_assignments ma ON ma.teacher_id = t.id AND ma.is_current = TRUE
LEFT JOIN projects p ON p.student_id = ma.student_id 
  AND p.status IN ('active', 'in_progress')
LEFT JOIN interventions i ON i.student_id = ma.student_id 
  AND i.status = 'pending'

WHERE t.status = 'active'

GROUP BY t.id, t.full_name, t.current_student_count, t.max_students

ORDER BY workload_percentage DESC;
```

### 测试 1：验证唯一性约束

```sql
-- 测试：不允许一个学生同时有两个当前班主任
DO $$
DECLARE
  v_student_id UUID := gen_random_uuid();
  v_teacher1_id UUID := gen_random_uuid();
  v_teacher2_id UUID := gen_random_uuid();
BEGIN
  -- 插入测试数据
  INSERT INTO students (id, user_id, nickname)
  VALUES (v_student_id, gen_random_uuid(), '测试学生');
  
  INSERT INTO teachers (id, user_id, full_name)
  VALUES 
    (v_teacher1_id, gen_random_uuid(), '班主任1'),
    (v_teacher2_id, gen_random_uuid(), '班主任2');
  
  -- 第一次分配：应该成功
  INSERT INTO mentor_assignments (student_id, teacher_id, is_current)
  VALUES (v_student_id, v_teacher1_id, TRUE);
  
  -- 第二次分配（不结束第一次）：应该失败
  BEGIN
    INSERT INTO mentor_assignments (student_id, teacher_id, is_current)
    VALUES (v_student_id, v_teacher2_id, TRUE);
    
    RAISE EXCEPTION '测试失败：允许了重复的当前班主任';
  EXCEPTION
    WHEN exclusion_violation THEN
      RAISE NOTICE '✅ 测试通过：唯一性约束生效';
  END;
  
  -- 清理
  ROLLBACK;
END $$;
```

### 2. 物化视图：班主任工作负载

```sql
CREATE MATERIALIZED VIEW teacher_workload_summary AS
SELECT 
  t.id AS teacher_id,
  t.full_name,
  t.current_student_count,
  t.max_students,
  COUNT(DISTINCT ma.student_id) AS verified_student_count,
  COUNT(DISTINCT p.id) AS active_projects_count,
  COUNT(DISTINCT i.id) AS pending_interventions_count,
  AVG(s.onboarding_completed::INT) AS onboarding_completion_rate
  
FROM teachers t
LEFT JOIN mentor_assignments ma ON ma.teacher_id = t.id AND ma.is_current = TRUE
LEFT JOIN students s ON s.id = ma.student_id
LEFT JOIN projects p ON p.student_id = s.id AND p.status IN ('active', 'in_progress')
LEFT JOIN interventions i ON i.student_id = s.id AND i.status = 'pending'

WHERE t.status = 'active'

GROUP BY t.id, t.full_name, t.current_student_count, t.max_students;

-- 创建索引
CREATE UNIQUE INDEX idx_teacher_workload_teacher 
  ON teacher_workload_summary(teacher_id);

-- 定时刷新（每小时）
CREATE OR REPLACE FUNCTION refresh_teacher_workload()
RETURNS void AS $$
BEGIN
  REFRESH MATERIALIZED VIEW CONCURRENTLY teacher_workload_summary;
END;
$$ LANGUAGE plpgsql;
```
